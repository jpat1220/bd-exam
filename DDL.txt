
CREATE TABLE tipo_usuario (
    tipo_usuario_id SERIAL PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL
);

CREATE TABLE ciudad (
    ciudad_id SERIAL PRIMARY KEY,
    nombre_ciudad VARCHAR(100) NOT NULL,
    codigo_postal VARCHAR(10)
);

CREATE TABLE usuario (
    usuario_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    tipo_usuario_id INT NOT NULL,
    condicion_especial VARCHAR(100),
    cargo VARCHAR(100),             -- Solo aplica si es empleado
    departamento VARCHAR(100),      -- Solo aplica si es empleado
    nit INT,   						-- Solo aplica si es proveedor	
    ciudad_id INT,
    FOREIGN KEY (tipo_usuario_id) REFERENCES tipo_usuario (tipo_usuario_id),
	FOREIGN KEY (ciudad_id) REFERENCES ciudad (ciudad_id)
);

CREATE TABLE servicio (
    servicio_id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT,
    estado VARCHAR(10) DEFAULT 'Activo' CHECK (estado IN ('Activo', 'Inactivo'))
);

CREATE TABLE turno (
    turno_id SERIAL PRIMARY KEY,
    usuario_id INT NOT NULL,     -- quien solicita el turno
    empleado_id INT NOT NULL,    -- quien atiende el turno
    servicio_id INT NOT NULL,
    fecha DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    hora_fin TIME NOT NULL,
    estado VARCHAR(15) DEFAULT 'Pendiente' CHECK (estado IN ('Pendiente', 'Atendido', 'Cancelado')),
    FOREIGN KEY (usuario_id) REFERENCES usuario(usuario_id),
    FOREIGN KEY (empleado_id) REFERENCES usuario(usuario_id),
    FOREIGN KEY (servicio_id) REFERENCES servicio(servicio_id)
);
